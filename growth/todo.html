目标：用 Docker Compose 起 Grafana + Prometheus + Redis + RabbitMQ + Kafka，学会用 Grafana 看这三个中间件。

对应：第11周 · Kafka 设计 + 可观测性入门

docker-compose 片段（未在本机运行验证）

# 未在本机运行验证；只含官方文档能确认的镜像和端口
services:
  grafana:
    image: grafana/grafana
    ports: ["3000:3000"]
  prometheus:
    image: prom/prometheus
    ports: ["9090:9090"]
    volumes: ["./prometheus.yml:/etc/prometheus/prometheus.yml"]
  redis:
    image: redis
    ports: ["6379:6379"]
  redis-exporter:
    image: oliver006/redis_exporter
    environment: ["REDIS_ADDR=redis://redis:6379"]
    ports: ["9121:9121"]
  rabbitmq:
    image: rabbitmq:4-management
    ports: ["5672:5672", "15672:15672", "15692:15692"]
  kafka:
    image: apache/kafka:4.4.0
    ports: ["9092:9092"]
    # 未核实：容器间访问需要的 listeners 配置
  kafka-exporter:
    image: danielqsj/kafka-exporter
    command: ["--kafka.server=kafka:9092"]
    ports: ["9308:9308"]
